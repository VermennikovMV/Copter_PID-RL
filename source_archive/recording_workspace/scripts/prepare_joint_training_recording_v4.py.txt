"""Prepare an exact fresh reproduction for manual recording; no learning by default.

Only --run launches the official preflight. --prepare-only creates and checks a
new plan/output namespace without importing the simulator, Torch or PPO.
"""
from __future__ import annotations
import argparse
import copy
from datetime import datetime, timezone
import json
from pathlib import Path
import subprocess
import sys
import uuid

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT))
from scripts import joint_research_preflight_v1 as gate

SOURCE = "docs/joint_corner_plans_v8/w8_control.bound.json"
SOURCE_SHA = "97a786dbbca6210c8eb949caa44fe752baded0a230fc1c5288acf0a01191560a"


def prepare():
    source = ROOT / SOURCE
    if gate.sha(source) != SOURCE_SHA:
        raise ValueError("Reviewed reproduction source plan changed")
    original = json.loads(source.read_text(encoding="utf-8-sig"))
    if original["variant"] != "w8_control" or original["budget"] != dict(ppo_steps=262144, critic_steps=131072, eval_every=131072, candidate_count=3):
        raise ValueError("Only the measured full-network short CONTROL recipe may be reproduced")
    stamp = datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%S%fZ")
    parent = ROOT / "artifacts/refinement/joint_v2" / ("recording_control77_" + stamp + "_" + uuid.uuid4().hex[:8])
    plan_path = parent / "research_plan.json"
    output = parent / "w8_control"
    plan = copy.deepcopy(original)
    argv = plan["argv"]
    argv[argv.index("--output") + 1] = output.relative_to(ROOT).as_posix()
    argv[argv.index("--research-plan") + 1] = plan_path.relative_to(ROOT).as_posix()
    plan["recording_reproduction"] = dict(
        title="Fresh reproduction of the measured short full-network CONTROL experiment",
        source_plan=dict(path=SOURCE, sha256=SOURCE_SHA), prepared_utc=datetime.now(timezone.utc).isoformat(),
        changed_only="argv --output and --research-plan paths, plus this recording annotation",
        initial_already_trained=True, independent_scientific_replication=False,
        viewed_development_seeds_reused=True, policy_acceptance=False,
        interpretation="Actual fresh PPO from the same trained initial actor; no replay, no from-scratch claim, no new independent confirmation or automatic Gazebo acceptance.")
    # Validate fresh output and every frozen prerequisite before writing anything.
    gate.validate(plan)
    recovered = copy.deepcopy(plan)
    recovered.pop("recording_reproduction")
    recovered["argv"] = list(original["argv"])
    if recovered != original:
        raise AssertionError("Reproduction changed scientific settings")
    parent.mkdir(exist_ok=False)
    with plan_path.open("x", encoding="utf-8") as stream:
        json.dump(plan, stream, ensure_ascii=False, indent=2)
        stream.write("\n")
    gate.validate(plan)
    return dict(status="PREPARED_NOT_TRAINED", plan=plan_path.relative_to(ROOT).as_posix(),
                plan_sha256=gate.sha(plan_path), output=output.relative_to(ROOT).as_posix(),
                source_plan_sha256=SOURCE_SHA, new_environment_steps=0, new_optimizer_steps=0)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    modes = parser.add_mutually_exclusive_group()
    modes.add_argument("--prepare-only", action="store_true", help="Prepare and validate only; starts no training")
    modes.add_argument("--run", action="store_true", help="User-invoked actual fresh learning through official preflight")
    args = parser.parse_args()
    receipt = prepare()
    print(json.dumps(receipt, ensure_ascii=False), flush=True)
    if args.run:
        return subprocess.run([sys.executable, "-B", str(ROOT / "scripts/joint_research_preflight_v1.py"),
                               "--plan", receipt["plan"], "--run"], cwd=ROOT, check=False).returncode
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
